package frc.robot.commands;

import frc.robot.subsystems.RotateSubsystem;

import java.util.ArrayList;

import edu.wpi.first.wpilibj2.command.Command;

/** An example command that uses an example subsystem. */
public class MinPowerCommand extends Command {
  private final RotateSubsystem rotate;

  enum State {
    START, COAST, MOVEFORWARD, MOVEBACK, DONE, MOVETONEXTPOINT
  }
  State state;
  int index;
  DataPoint dataPoint;
  ArrayList<DataPoint> dataPoints = new ArrayList<>();
  State prevState;
  boolean firstTime;
  long endTime;
  boolean moveForward;
  int pointsDone;
  double maxPos = .21;
  double pointDelta = .015;
  double numberOfPoints = 0;
  boolean done;
  double nextPointPos;
  double shuffleForward;
  double shuffleBack;
  double shuffleSpeedInc = .01;
  boolean notDoneAlready;


  public MinPowerCommand(RotateSubsystem rotate, boolean moveForward) {
    this.rotate = rotate;
    this.moveForward = moveForward;
    // Use addRequirements() here to declare subsystem dependencies.
  }

  // Called when the command is initially scheduled.
  @Override
  public void initialize() {
    rotate.resetSafe();
    dataPoints.clear();
    state = State.START;
    prevState = State.DONE; // any non START state
    firstTime = true;
    numberOfPoints = 1;
    done = false;
  }

  // Called every time the scheduler runs while the command is scheduled.
  @Override
  public void execute() {
    firstTime = prevState.equals(state);

    switch (state){
      case START:
        dataPoint = new DataPoint();
        dataPoints.add(dataPoint);
        dataPoint.startPos = rotate.getLeftRotateEncoder();
        dataPoint.startVel = rotate.getLeftRotateEncoderVel();
        rotate.setRotationSpeed(0, 0);
        state = State.COAST;
        break;
      case COAST:
        if(firstTime){
          endTime = System.currentTimeMillis()+500; // half second
        }
        else if(System.currentTimeMillis() > endTime){
          notDoneAlready = false;
          dataPoint.coastPos = rotate.getLeftRotateEncoder();
          dataPoint.coastVel = rotate.getLeftRotateEncoderVel();
          if(moveForward){
            state = State.MOVEFORWARD;
          }
          else{
            state = State.MOVEBACK;
          }
        }
        break;
      case MOVEFORWARD:
        if (firstTime){
          nextPointPos = dataPoint.startPos + numberOfPoints*pointDelta;
          shuffleForward = nextPointPos + pointDelta;
          shuffleBack = nextPointPos - pointDelta;
          if (shuffleForward > maxPos || shuffleBack<0){
            done = true;
          }
        }
        


        break;
      case MOVEBACK:
        break;
      case DONE:
        break;
      case MOVETONEXTPOINT:
        break;
    }
  }

  // Called once the command ends or is interrupted.
  @Override
  public void end(boolean interrupted) {}

  // Returns true when the command should end.
  @Override
  public boolean isFinished() {
    return done;
  }
}

class DataPoint{
  double startPos;
  double startVel;
  double coastPos;
  double coastVel;
  double minPowerForward;
  double minPowerBack;

  public DataPoint(){

  }
}
